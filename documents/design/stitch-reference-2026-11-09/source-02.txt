  (function() {
    let seconds = 4;
    let isRecording = true;
    const timerEl = document.getElementById('recording-timer');
    const toggleBtn = document.getElementById('btn-toggle-rec');
    const waveform = document.getElementById('waveform-container');

    const interval = setInterval(() => {
      if (isRecording) {
        seconds++;
        const mins = String(Math.floor(seconds / 60)).padStart(2, '0');

    if (toggleBtn) {
      toggleBtn.addEventListener('click', () => {

        if (!isRecording) {
          toggleBtn.classList.remove('bg-error', 'text-on-error');
          toggleBtn.classList.add('bg-primary', 'text-on-primary');
          toggleBtn.innerHTML = '<span class="material-symbols-outlined text-[34px]" style="font-variation-settings: \'FILL\' 1;">mic</span>';

          toggleBtn.classList.remove('bg-primary', 'text-on-primary');
          toggleBtn.classList.add('bg-error', 'text-on-error');
          toggleBtn.innerHTML = '<span class="material-symbols-outlined text-[34px]" style="font-variation-settings: \'FILL\' 1;">stop</span>';
          if (waveform) waveform.classList.remove('opacity-40');
        }
      });
    }

    const restartBtn = document.getElementById('btn-restart');
    if (restartBtn) {
      restartBtn.addEventListener('click', () => {
        seconds = 0;
        if (timerEl) timerEl.textContent = '00:00';
      });
    }
  })();

  function handleTyping(textarea) {
    const len = textarea.value.length;
    const charCountEl = document.getElementById('char-count');
    const progressEl = document.getElementById('typing-progress');
    
    if (charCountEl) {
      charCountEl.innerText = len + ' ký tự';

      progressEl.style.width = pct + '%';
    }

  function toggleHintAccordion() {
    const box = document.getElementById('hint-box');
    const icon = document.getElementById('hint-icon');
    const btn = document.getElementById('hint-toggle-btn');
    const isHidden = box.classList.contains('hidden');
    
    if (isHidden) {
      box.classList.remove('hidden');
      icon.style.transform = 'rotate(180deg)';
      btn.setAttribute('aria-expanded', 'true');
    } else {
      box.classList.add('hidden');
      icon.style.transform = 'rotate(0deg)';
      btn.setAttribute('aria-expanded', 'false');
    }
  }

  function toggleRomajiHelp() {
    const label = document.getElementById('label-romaji');

    } else {
      label.innerText = 'Bật gõ Romaji';
    }
  }

  function switchInputMode(mode) {
    const wbBtn = document.getElementById('mode-wordbank');
    const kbBtn = document.getElementById('mode-keyboard');
    const wbSection = document.getElementById('wordbank-section');
    const kbSection = document.getElementById('keyboard-section');
    const imeInput = document.getElementById('ime-input');

    if (mode === 'wordbank') {
      wbBtn.className = 'flex-1 flex items-center justify-center gap-1.5 py-2.5 rounded-lg bg-surface-container-lowest text-primary font-label-md text-label-md shadow-sm transition-all';
      kbBtn.className = 'flex-1 flex items-center justify-center gap-1.5 py-2.5 rounded-lg text-secondary hover:text-on-surface font-label-md text-label-md transition-all';
      wbSection.classList.remove('hidden');
      kbSection.classList.add('hidden');
    } else {
      kbBtn.className = 'flex-1 flex items-center justify-center gap-1.5 py-2.5 rounded-lg bg-surface-container-lowest text-primary font-label-md text-label-md shadow-sm transition-all';
      wbBtn.className = 'flex-1 flex items-center justify-center gap-1.5 py-2.5 rounded-lg text-secondary hover:text-on-surface font-label-md text-label-md transition-all';
      wbSection.classList.add('hidden');
      kbSection.classList.remove('hidden');
      imeInput.focus();
    }
  }

  function selectOption(btn, val) {
    const allChips = document.querySelectorAll('.bank-chip');
    allChips.forEach(chip => {
      chip.className = 'bank-chip h-12 px-space-md rounded-xl font-body-japanese text-body-japanese flex items-center justify-center gap-2 bg-surface-container-lowest text-on-surface shadow-sm hover:bg-surface-container active:scale-95 transition-all';
      chip.querySelector('.chip-icon').classList.add('hidden');
    });

    btn.className = 'bank-chip h-12 px-space-md rounded-xl font-body-japanese text-body-japanese flex items-center justify-center gap-2 bg-primary-container text-on-primary shadow-sm active:scale-95 transition-all';
    btn.querySelector('.chip-icon').classList.remove('hidden');

    const blankValue = document.getElementById('blank-value');
    blankValue.textContent = val;
  }

  function handleImeInput(val) {
    const blankValue = document.getElementById('blank-value');
    blankValue.textContent = val.trim() || '_____';
  }

  function clearImeInput() {
    const imeInput = document.getElementById('ime-input');
    imeInput.value = '';
    handleImeInput('');
    imeInput.focus();
  }
</script></main>
