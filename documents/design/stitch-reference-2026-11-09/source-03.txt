  // Dynamic Character Counter & Edit Tracker
  const textarea = document.getElementById('japanese-text-input');
  const counter = document.getElementById('char-count-badge');
  const revertBtn = document.getElementById('revert-btn');
  const originalText = '日本へ行ったことがあります。';

  function updateCharCount() {
    if (!textarea) return;
    const count = textarea.value.trim().length;
    counter.textContent = `${count} ký tự`;
    
    if (textarea.value !== originalText) {
      revertBtn.classList.remove('hidden');
      revertBtn.classList.add('inline-flex');
    } else {
      revertBtn.classList.add('hidden');
      revertBtn.classList.remove('inline-flex');
    }
  }

  textarea.addEventListener('input', updateCharCount);

  if (revertBtn) {
    revertBtn.addEventListener('click', () => {
      textarea.value = originalText;
      updateCharCount();
    });
  }

  // Japanese punctuation insert helper
  function insertCharacter(char) {
    if (!textarea) return;
    const start = textarea.selectionStart;
    const end = textarea.selectionEnd;
    const text = textarea.value;
    textarea.value = text.substring(0, start) + char + text.substring(end);
    textarea.selectionStart = textarea.selectionEnd = start + char.length;
    textarea.focus();
    updateCharCount();
  }

  // Clear for manual input mode
  function clearForManualInput() {
    if (!textarea) return;
    textarea.value = '';
    textarea.focus();
    updateCharCount();
  }

  // Interactive Token Selector Toggle
  const tokenButtons = document.querySelectorAll('.token-btn');
  tokenButtons.forEach(btn => {
    btn.addEventListener('click', () => {
      const isActive = btn.getAttribute('data-active') === 'true';
      if (isActive) {
        btn.setAttribute('data-active', 'false');
        btn.classList.remove('bg-secondary-container', 'text-on-primary-fixed-variant', 'font-semibold', 'shadow-xs');
        btn.classList.add('bg-surface-container', 'text-on-surface');
      } else {
        btn.setAttribute('data-active', 'true');
        btn.classList.remove('bg-surface-container', 'text-on-surface');
        btn.classList.add('bg-secondary-container', 'text-on-primary-fixed-variant', 'font-semibold', 'shadow-xs');
      }
    });
  });
