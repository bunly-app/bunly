            clearInterval(interval);
            isRetrying = false;
            btn.disabled = false;

            icon.classList.remove('animate-spin');
            text.textContent = 'Thử kết nối lại';

  (function() {
    function setupLoadingButton(btnId, loadingText, successText, duration) {
      var btn = document.getElementById(btnId);
      if (!btn) return;
      var textEl = btn.querySelector('.btn-text');
      var spinnerEl = btn.querySelector('.spinner');
      var checkEl = btn.querySelector('.check-icon');
      var originalText = textEl ? textEl.textContent : '';

      btn.addEventListener('click', function() {
        if (btn.getAttribute('aria-busy') === 'true') return;

        // Lock button, prevent double submission
        btn.setAttribute('aria-busy', 'true');
        btn.classList.add('opacity-90', 'pointer-events-none');
        if (spinnerEl) spinnerEl.classList.remove('hidden');
        if (checkEl) checkEl.classList.add('hidden');
        if (textEl) textEl.textContent = loadingText;

        setTimeout(function() {
          if (spinnerEl) spinnerEl.classList.add('hidden');
          if (checkEl) checkEl.classList.remove('hidden');
          if (textEl) textEl.textContent = successText || originalText;

          setTimeout(function() {
            btn.removeAttribute('aria-busy');
            btn.classList.remove('opacity-90', 'pointer-events-none');
            if (checkEl) checkEl.classList.add('hidden');
            if (textEl) textEl.textContent = originalText;
          }, 1400);
        }, duration || 1800);
      });
    }

    setupLoadingButton('btn-login-demo', 'Đang đăng nhập…', 'Đăng nhập thành công', 1700);
    setupLoadingButton('btn-create-demo', 'Đang tạo tài khoản…', 'Tài khoản đã tạo', 1900);
    setupLoadingButton('btn-submit-demo', 'Đang gửi câu để phân tích…', 'Phân tích hoàn tất', 2200);
    setupLoadingButton('btn-save-demo', 'Đang lưu trên thiết bị…', 'Đã lưu trên thiết bị', 1600);

    // Toggle Skeleton Preview
    var toggleBtn = document.getElementById('btn-toggle-skeleton');
    var skeletonCard = document.getElementById('skeleton-card');
    var ghostEl = document.getElementById('skeleton-ghost');
    var realEl = document.getElementById('skeleton-real');

    if (toggleBtn && skeletonCard && ghostEl && realEl) {
      toggleBtn.addEventListener('click', function() {
        var isBusy = skeletonCard.getAttribute('aria-busy') === 'true';
        if (isBusy) {
          skeletonCard.setAttribute('aria-busy', 'false');
          ghostEl.classList.add('hidden');
          realEl.classList.remove('hidden');
          toggleBtn.textContent = 'Xem Skeleton';
        } else {
          skeletonCard.setAttribute('aria-busy', 'true');
          ghostEl.classList.remove('hidden');
          realEl.classList.add('hidden');
          toggleBtn.textContent = 'Xem Nội dung';
        }
      });
