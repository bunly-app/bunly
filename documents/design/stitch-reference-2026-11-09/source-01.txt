<script>
  (function() {
    // 1. Chuyển hướng khi click vào hàng ngữ pháp sang màn chi tiết S08
    const grammarRows = document.querySelectorAll('.grammar-row');
    grammarRows.forEach(row => {
      row.addEventListener('click', (e) => {
        // Tránh kích hoạt khi bấm nút bookmark
        if (e.target.closest('.bookmark-btn')) return;
        const grammarId = row.dataset.id;
        // Kích hoạt điều hướng hoặc mô phỏng mở chi tiết mẫu ngữ pháp
        const targetUrl = `#chi-tiet-ngu-phap?id=${encodeURIComponent(grammarId)}`;
        window.location.hash = targetUrl;
      });
    });

    // 2. Chuyển đổi trạng thái Bookmark nhẹ nhàng (Bookmark toggle)
    const bookmarkButtons = document.querySelectorAll('.bookmark-btn');
    bookmarkButtons.forEach(btn => {
      btn.addEventListener('click', (e) => {
        e.stopPropagation();
        const isSaved = btn.getAttribute('data-saved') === 'true';
        const icon = btn.querySelector('.material-symbols-outlined');
        
        if (isSaved) {
          btn.setAttribute('data-saved', 'false');
          btn.classList.remove('text-primary');
          btn.classList.add('text-outline');
          icon.textContent = 'bookmark_border';
          icon.style.fontVariationSettings = "'FILL' 0";
        } else {
          btn.setAttribute('data-saved', 'true');
          btn.classList.add('text-primary');
          btn.classList.remove('text-outline');
          icon.textContent = 'bookmark';
          icon.style.fontVariationSettings = "'FILL' 1";
        }
      });
    });

    // 3. Xử lý Input Search & nút xóa
    const searchInput = document.getElementById('grammar-search');
    const clearBtn = document.getElementById('clear-search');

    if (searchInput && clearBtn) {
      searchInput.addEventListener('input', () => {
        const val = searchInput.value.trim().toLowerCase();
        clearBtn.classList.toggle('hidden', val.length === 0);

        // Lọc danh sách ngữ pháp trực quan tại chỗ
        grammarRows.forEach(row => {
          const textContent = row.textContent.toLowerCase();
          if (val === '' || textContent.includes(val)) {
            row.style.display = 'flex';
          } else {
            row.style.display = 'none';
          }
        });
      });

      clearBtn.addEventListener('click', () => {
        searchInput.value = '';
        clearBtn.classList.add('hidden');
        grammarRows.forEach(row => row.style.display = 'flex');
        searchInput.focus();
      });
    }

    // 4. Toggle hiệu ứng tương tác các Chip lọc
